# Same payload-file hygiene as the wildcard group: a named group replaces the
# wildcard entirely for its agent, so the rules must be repeated here.
Disallow: /api/

Disallow: /index.txt
Disallow: /*/index.txt
Disallow: /__next
Disallow: /*/__next

# The legal pages (terms, privacy, cookies) used to be disallowed here. That
# was the wrong instrument: they carry a noindex tag, and a crawler that is
# not allowed to fetch them never sees it. Search Console then reports them as
# "blocked by robots.txt" indefinitely, and the URLs can still surface without
# a snippet. Crawling is allowed so the noindex can be read and obeyed.
User-agent: *
Allow: /
Disallow: /api/

# Next.js static-export payload files. The browser needs them for client-side
# navigation, so they stay reachable, but crawling them is pure waste: they
# duplicate page content as plain text and Googlebot only makes about forty
# content requests a fortnight on this domain.
Disallow: /*_rsc=
Disallow: /index.txt
Disallow: /*/index.txt
Disallow: /__next
Disallow: /*/__next
